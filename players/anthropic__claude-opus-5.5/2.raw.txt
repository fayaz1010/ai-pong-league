function createPlayer() {
  const W = 800, H = 500, FACE = 30, OFACE = 770, PH = 40, SPD = 5;
  let planOff = null, say = "Warming up. Let's play some Pong.", lastDir = 0;
  let lastScore = "0-0", sayLock = 0, cat = "";

  function clamp(v, a, b) { return v < a ? a : v > b ? b : v; }

  function reflect(y, r) {
    const L = H - 2 * r;
    let v = (y - r) % (2 * L);
    if (v < 0) v += 2 * L;
    if (v > L) v = 2 * L - v;
    return v + r;
  }

  // analytic prediction (used for hypothetical shots)
  function predA(x, y, dx, dy, tx, r) {
    const t = (tx - x) / dx;
    return { y: reflect(y + dy * t, r), t: t };
  }

  // frame-by-frame prediction of where ball meets a face
  function predD(x, y, dx, dy, r, left) {
    let n = 0;
    while (n < 2000) {
      if (left ? (x - r <= FACE) : (x + r >= OFACE)) break;
      x += dx; y += dy;
      if (y < r) { y = 2 * r - y; dy = -dy; }
      else if (y > H - r) { y = 2 * (H - r) - y; dy = -dy; }
      n++;
    }
    return { y: y, t: n };
  }

  function setSay(text, c) {
    if (c !== cat && sayLock <= 0) { say = text.slice(0, 48); cat = c; }
  }

  return function move(s) {
    const b = s.ball, me = s.me.y, op = s.opp.y;
    const r = b.r || 6;
    const speed = Math.hypot(b.dx, b.dy);
    const dir = b.dx < 0 ? -1 : 1;
    if (sayLock > 0) sayLock--;

    const sc = s.score.me + "-" + s.score.opp;
    if (sc !== lastScore) {
      const won = s.score.me > +lastScore.split("-")[0];
      lastScore = sc;
      say = (won ? "Point to me! " : "Ouch, they got one. ") + sc;
      cat = "score"; sayLock = 90;
    }
    if (dir !== lastDir) { planOff = null; lastDir = dir; }

    let targetPaddle = H / 2, meetY = H / 2;

    if (b.dx < 0) {
      // incoming: find where it lands, choose the nastiest reachable return
      const p = predD(b.x, b.y, b.dx, b.dy, r, true);
      const n = Math.max(0, p.t - 1);
      const reach = SPD * n + 1;
      const ns = Math.min(speed * 1.05, 24);
      const maxOff = ns > 18 ? 0.8 : 0.88;
      let best = null;
      for (let off = -maxOff; off <= maxOff + 1e-9; off += 0.04) {
        const pc = p.y - off * PH;
        if (pc < PH || pc > H - PH) continue;
        if (Math.abs(pc - me) > reach) continue;
        const a = off * 55 * Math.PI / 180;
        const q = predA(FACE + r, p.y, ns * Math.cos(a), ns * Math.sin(a), OFACE - r, r);
        const need = Math.max(0, Math.abs(q.y - op) - PH - r + 4);
        let score = need - SPD * q.t - 0.3 * q.t;
        if (planOff !== null && Math.abs(off - planOff) < 0.05) score += 6;
        if (!best || score > best.score) best = { off, pc, score, qy: q.y, t: q.t };
      }
      meetY = p.y;
      if (best) {
        planOff = best.off;
        targetPaddle = best.pc;
        const where = best.qy < H / 3 ? "top" : best.qy > 2 * H / 3 ? "bottom" : "middle";
        if (best.score > 0) {
          setSay("Angling it " + where + " - they can't get there!", "kill" + where);
        } else if (Math.abs(best.off) > 0.5) {
          setSay("Sharp angle to the " + where + ", make 'em run", "ang" + where);
        } else {
          setSay("Safe return, speed it up " + where, "safe" + where);
        }
      } else {
        targetPaddle = clamp(p.y, PH, H - PH);
        setSay("Lunging for it! Full stretch!", "lunge");
      }
    } else {
      // outgoing: predict their hit, cover the midpoint of all their returns
      const p = predD(b.x, b.y, b.dx, b.dy, r, false);
      const ns = Math.min(speed * 1.05, 24);
      let lo = H, hi = 0;
      for (let off = -1; off <= 1.0001; off += 0.1) {
        const a = off * 55 * Math.PI / 180;
        const q = predA(OFACE - r, p.y, -ns * Math.cos(a), ns * Math.sin(a), FACE + r, r);
        if (q.y < lo) lo = q.y;
        if (q.y > hi) hi = q.y;
      }
      targetPaddle = clamp((lo + hi) / 2, PH, H - PH);
      meetY = targetPaddle;
      setSay("Covering every return they've got", "cover");
    }

    const mv = clamp((targetPaddle - me) / SPD, -1, 1);
    return { move: mv, say: say, target: meetY };
  };
}
